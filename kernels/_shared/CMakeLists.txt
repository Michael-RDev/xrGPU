cmake_minimum_required(VERSION 3.24)
project(xrgpu_native_operators LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(XRGPU_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(xrgpu_operator_plan STATIC planner.cpp)
target_include_directories(xrgpu_operator_plan PUBLIC "${XRGPU_ROOT}/runtime/include")
option(XRGPU_OPERATORS_PLAN_ONLY "Build metadata planner without Vulkan headers" OFF)
option(XRGPU_OPERATORS_BUILD_LEGACY_RUNTIME "Build the original native Vulkan runtime" ON)
option(XRGPU_OPERATORS_TEST_SOFTWARE_VULKAN "Explicit test-only software Vulkan runtime" OFF)
if(XRGPU_OPERATORS_TEST_SOFTWARE_VULKAN AND ANDROID)
  message(FATAL_ERROR "Software Vulkan test override cannot be built into Android production artifacts")
endif()
if(NOT XRGPU_OPERATORS_PLAN_ONLY)
  set(XRGPU_OPERATOR_GENERATED_DIR "${CMAKE_CURRENT_SOURCE_DIR}/generated" CACHE PATH "Inventoried shader hashes")
  if(NOT ANDROID)
    find_path(XRGPU_VULKAN_INCLUDE_DIR vulkan/vulkan.h REQUIRED)
  endif()
  if(XRGPU_OPERATORS_BUILD_LEGACY_RUNTIME)
    if(NOT EXISTS "${XRGPU_OPERATOR_GENERATED_DIR}/shader_hashes.h")
      message(FATAL_ERROR "Use python -m kernels._shared.build to compile and inventory native shaders")
    endif()
    add_library(xrgpu_operators SHARED vulkan.cpp)
    if(XRGPU_OPERATORS_TEST_SOFTWARE_VULKAN)
      target_compile_definitions(xrgpu_operators PRIVATE XRGPU_OPERATOR_TEST_SOFTWARE_VULKAN=1)
      find_path(XRGPU_JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
      add_executable(xrgpu_operator_execute tests/execute.cpp)
      target_include_directories(xrgpu_operator_execute PRIVATE "${XRGPU_JSON_INCLUDE_DIR}")
      target_link_libraries(xrgpu_operator_execute PRIVATE xrgpu_operators)
    endif()
    target_include_directories(xrgpu_operators PRIVATE "${XRGPU_OPERATOR_GENERATED_DIR}")
    target_link_libraries(xrgpu_operators PRIVATE xrgpu_operator_plan ${CMAKE_DL_LIBS})
    if(ANDROID)
      target_link_options(xrgpu_operators PRIVATE "-Wl,-z,max-page-size=16384")
    else()
      target_include_directories(xrgpu_operators PRIVATE "${XRGPU_VULKAN_INCLUDE_DIR}")
    endif()
    target_include_directories(xrgpu_operators PUBLIC "${XRGPU_ROOT}/runtime/include")
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(xrgpu_operator_plan_tests tests/planner.cpp)
  target_link_libraries(xrgpu_operator_plan_tests PRIVATE xrgpu_operator_plan)
  add_test(NAME operator_plan COMMAND xrgpu_operator_plan_tests)
endif()

# Generic typed N-input/N-output kernels remain separate from the legacy ABI.
option(XRGPU_OPERATORS_BUILD_EXTENDED "Build generic native operator families" ON)
if(XRGPU_OPERATORS_BUILD_EXTENDED)
  file(GLOB XRGPU_EXTENDED_FAMILIES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/extended/*_plan.cpp")
  add_library(xrgpu_extended_plan STATIC extended/planner.cpp extended/contracts.cpp ${XRGPU_EXTENDED_FAMILIES})
  target_include_directories(xrgpu_extended_plan PUBLIC "${XRGPU_ROOT}/runtime/include")
  target_link_libraries(xrgpu_extended_plan PUBLIC xrgpu_operator_plan)
  if(NOT XRGPU_OPERATORS_PLAN_ONLY)
    if(NOT EXISTS "${XRGPU_OPERATOR_GENERATED_DIR}/extended_shader_hashes.h")
      message(FATAL_ERROR "The checked native catalog's compiled shader inventory is required")
    endif()
    add_library(xrgpu_extended_operators SHARED extended/vulkan.cpp)
    target_include_directories(xrgpu_extended_operators PUBLIC "${XRGPU_ROOT}/runtime/include" PRIVATE "${XRGPU_OPERATOR_GENERATED_DIR}")
    target_link_libraries(xrgpu_extended_operators PRIVATE xrgpu_extended_plan ${CMAKE_DL_LIBS})
    if(XRGPU_OPERATORS_TEST_SOFTWARE_VULKAN)
      target_compile_definitions(xrgpu_extended_operators PRIVATE XRGPU_OPERATOR_TEST_SOFTWARE_VULKAN=1)
    endif()
    if(ANDROID)
      target_link_options(xrgpu_extended_operators PRIVATE "-Wl,-z,max-page-size=16384")
    else()
      target_include_directories(xrgpu_extended_operators PRIVATE "${XRGPU_VULKAN_INCLUDE_DIR}")
    endif()
  endif()
  if(BUILD_TESTING)
    add_executable(xrgpu_extended_framework_tests extended/tests/framework.cpp)
    target_link_libraries(xrgpu_extended_framework_tests PRIVATE xrgpu_extended_plan)
    add_test(NAME extended_framework COMMAND xrgpu_extended_framework_tests)
  endif()
  option(XRGPU_OPERATORS_BUILD_EXTENDED_TEST_RUNNER "Build JSON test adapter (not part of production API)" OFF)
  if(XRGPU_OPERATORS_BUILD_EXTENDED_TEST_RUNNER)
    find_path(XRGPU_JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
    add_executable(xrgpu_extended_execute extended/tests/execute.cpp)
    target_include_directories(xrgpu_extended_execute PRIVATE "${XRGPU_JSON_INCLUDE_DIR}")
    if(XRGPU_OPERATORS_PLAN_ONLY)
      target_compile_definitions(xrgpu_extended_execute PRIVATE XRGPU_EXTENDED_PLAN_ONLY=1)
      target_link_libraries(xrgpu_extended_execute PRIVATE xrgpu_extended_plan)
    else()
      target_link_libraries(xrgpu_extended_execute PRIVATE xrgpu_extended_operators xrgpu_extended_plan)
    endif()
  endif()
endif()
